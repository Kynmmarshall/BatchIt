%% SLIDE 5 — OOP CONCEPTS
\begin{frame}{Object-Oriented Design --- Concepts \& Patterns Applied}
\vspace{0.06cm}
\begin{columns}[T]
\begin{column}{0.49\textwidth}

  \begin{tcolorbox}[cardbox,title={Encapsulation}]
    \scriptsize Data and behaviour bundled inside model classes.
    \texttt{Batch} exposes only clean methods to callers:
    \begin{itemize}\setlength\itemsep{1pt}
      \item \texttt{progress\_ratio()} --- computes fill percentage
      \item \texttt{can\_join(qty)} --- validates remaining capacity
    \end{itemize}
    Flutter services (\texttt{BatchService}, \texttt{ApiClient}) hide all HTTP internals from UI widgets.
  \end{tcolorbox}

  \vspace{-0.12cm}

  \begin{tcolorbox}[cardbox,title={Inheritance}]
    \scriptsize Django \texttt{AbstractUser} extended by two concrete user types:
    \begin{itemize}\setlength\itemsep{1pt}
      \item \texttt{Customer} --- phone, preferences, profile photo
      \item \texttt{Provider} --- business name, GPS coords, verification status
    \end{itemize}
    Flutter: \texttt{StatefulWidget} / \texttt{StatelessWidget} base classes inherited across all 20+ screens.
  \end{tcolorbox}

\end{column}
\begin{column}{0.49\textwidth}

  \begin{tcolorbox}[orangecard,title={Polymorphism}]
    \scriptsize Same interface, different runtime behaviour:
    \begin{itemize}\setlength\itemsep{1pt}
      \item \texttt{BasePermission} $\to$ \texttt{IsCustomer} / \texttt{IsProvider} / \texttt{IsAdmin}
      \item \texttt{ModelSerializer} $\to$ \texttt{BatchSerializer} / \texttt{OrderSerializer}
    \end{itemize}
  \end{tcolorbox}

  \vspace{0.16cm}

  \begin{tcolorbox}[orangecard,title={Abstraction \& Composition}]
    \scriptsize
    \textbf{Abstraction:} Service layer hides REST API details behind typed methods (\texttt{fetchBatches()}, \texttt{joinBatch()}).\\[3pt]
    \textbf{Composition:} \texttt{Batch} owns one \texttt{BatchChatRoom} (1:1) and many \texttt{BatchParticipant} objects (1:N). \texttt{Customer} relates to \texttt{Provider} via a \texttt{Subscription} join object.
  \end{tcolorbox}

\end{column}
\end{columns}

\vspace{0.16cm}

\begin{tcolorbox}[tealstat,title={Design Patterns}]
  \scriptsize
  \textbf{Singleton} --- \texttt{ApiClient} single shared instance \quad\textbullet\quad
  \textbf{Repository} --- Service classes decouple UI from data \quad\textbullet\quad
  \textbf{Observer} --- Flutter \texttt{ChangeNotifier} / \texttt{Provider} for reactive state \quad\textbullet\quad
  \textbf{Strategy} --- Swappable DRF permission and serializer classes
\end{tcolorbox}

\end{frame}

%% SLIDE 6 — DESIGN PATTERNS
\begin{frame}{Design Patterns Applied}
\vspace{0.06cm}
\begin{columns}[T]
\begin{column}{0.49\textwidth}

  \begin{tcolorbox}[cardbox,title={Creational Patterns (Flutter)}]
    \scriptsize
    \textbf{Singleton} --- \texttt{ApiClient} uses a private constructor and a static \texttt{\_instance}; all six service classes share one HTTP client and one auth-token state.\\[3pt]
    \textbf{Factory} --- Every model class exposes a \texttt{fromJson()} factory constructor; the service layer instantiates them directly from decoded API responses.
  \end{tcolorbox}

  \vspace{0.12cm}

  \begin{tcolorbox}[cardbox,title={Structural Patterns (Flutter)}]
    \scriptsize
    \textbf{Facade} --- \texttt{ApiClient} hides raw \texttt{http.Client} calls; token injection, 401 retry, and error parsing are invisible to every caller.\\[3pt]
    \textbf{Repository} --- \texttt{BatchService}, \texttt{AuthService}, and \texttt{ProviderService} isolate all HTTP details from the six \texttt{ChangeNotifier} providers.
  \end{tcolorbox}

\end{column}
\begin{column}{0.49\textwidth}

  \begin{tcolorbox}[orangecard,title={Behavioural Patterns}]
    \scriptsize
    \textbf{Observer} \textit{(Flutter)} --- Six \texttt{ChangeNotifier} subclasses drive reactive UI: \texttt{notifyListeners()} triggers \texttt{Consumer} and \texttt{context.watch()} rebuilds automatically.\\[3pt]
    \textbf{Strategy} \textit{(Django)} --- DRF permission classes (\texttt{IsCustomer}, \texttt{IsProvider}, \texttt{IsAdmin}) and \texttt{ModelSerializer} subclasses are selected per endpoint at runtime.
  \end{tcolorbox}

  \vspace{0.12cm}

  \begin{tcolorbox}[orangecard,title={Architectural Patterns (Django)}]
    \scriptsize
    \textbf{MVT} --- \texttt{models.py} defines schema; DRF \texttt{ListCreateAPIView} and \texttt{RetrieveUpdateDestroyAPIView} handle all REST logic without templates.\\[3pt]
    \textbf{Serializer / Adapter} --- DRF serialisers map DB field names to the public API contract (e.g.\ \texttt{total\_quantity} $\to$ \texttt{bulk\_size\_kg}).\\[3pt]
    \textbf{Transaction Script} --- \texttt{@transaction.atomic} keeps multi-step batch creation consistent under concurrent writes.
  \end{tcolorbox}

\end{column}
\end{columns}

\vspace{0.10cm}

\begin{tcolorbox}[tealstat,title={Separation of Concerns --- Runtime Data Flow}]
  \scriptsize
  \textbf{Widget} $\to$ \texttt{Provider} (\textit{Observer}) $\to$ \texttt{Service} (\textit{Repository}) $\to$ \texttt{ApiClient} (\textit{Facade/Singleton}) $\to$ \textbf{HTTP} $\to$ \texttt{Django View} (\textit{MVT}) $\to$ \texttt{Serialiser} (\textit{Adapter}) $\to$ \textbf{DB}
\end{tcolorbox}
